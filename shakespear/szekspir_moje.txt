    "# Small LSTM Model\n",
    "class LSTM(nn.Module):\n",
    "    def __init__(self, n_hidden = 256, n_layers = 3, drop_prob = 0.5):\n",
    "        super(LSTM, self).__init__()\n",
    "        self.n_hidden = n_hidden\n",
    "        self.n_layers = n_layers\n",
    "    \n",
    "        self.embedding = nn.Embedding(n_chars, n_hidden)\n",
    "        self.lstm = nn.LSTM(n_hidden, n_hidden, n_layers, dropout = drop_prob, batch_first = True)\n",
    "        self.dropout = nn.Dropout(drop_prob)\n",
    "        self.fc = nn.Linear(n_hidden, n_chars)\n",
    "\n",
    "    def forward(self, x, hidden):\n",
    "        x = self.embedding(x)\n",
    "        r_output, hidden = self.lstm(x, hidden)\n",
    "        out = self.dropout(r_output)\n",
    "        out = out.contiguous().view(-1, self.n_hidden)\n",
    "        out = self.fc(out)\n",
    "        return out, hidden\n",
    "\n",
    "    def init_hidden(self, batch_size):\n",
    "        weight = next(self.parameters()).data\n",
    "        hidden = (weight.new(self.n_layers, batch_size, self.n_hidden).zero_(), weight.new(self.n_layers, batch_size, self.n_hidden).zero_())\n",
    "        return hidden"
   ]

    "def generate_text(model, start_sequence, char_to_idx, idx_to_char, generation_length=100, temperature=1.0):\n",
    "    model.eval()\n",
    "    \n",
    "    input_sequence = [char2int[char] for char in start_sequence]\n",
    "    input_sequence = torch.tensor(input_sequence, dtype=torch.long).unsqueeze(0)  # Adding one dimension (1, seq_length)\n",
    "    \n",
    "    hidden = model.init_hidden(1)  # Batch size = 1\n",
    "    \n",
    "    generated_text = start_sequence\n",
    "    \n",
    "    for i in range(len(start_sequence) - 1):\n",
    "        output, hidden = model(input_sequence[:, i].unsqueeze(0), hidden)\n",
    "    \n",
    "    input_char = input_sequence[:, -1]\n",
    "    \n",
    "    for _ in range(generation_length):\n",
    "        output, hidden = model(input_char.unsqueeze(0), hidden)\n",
    "        output = output.squeeze().div(temperature).exp()\n",
    "        probabilities = torch.multinomial(output, 1).item()\n",
    "        \n",
    "        next_char = idx_to_char[probabilities]\n",
    "        generated_text += next_char\n",
    "        \n",
    "        input_char = torch.tensor([probabilities], dtype=torch.long)\n",
    "    \n",
    "    return generated_text"
   ]
